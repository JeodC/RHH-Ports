set -e

# Z3R is built from the latest stable upstream release. We intentionally build
# only the executable target so CI never extracts or redistributes copyrighted
# game assets. Users supply their own zelda3_assets.dat.
git clone --depth 1 --branch v0.2.4 https://github.com/xander-haj/Z3R.git Z3R
cd Z3R

# Build only the native executable; the upstream `all` target also runs
# assets/restool.py --extract-from-rom, which is deliberately not used here.
make -j$(nproc) zelda3

# Keep the shipped configuration alongside the executable, forcing the
# software-independent SDL output backend for the RG35XX-H.
sed 's/^OutputMethod = .*/OutputMethod = SDL/' zelda3.ini > ../zelda3.ini

# Verify the expected executable and SDL configuration were produced.
test -x zelda3
grep -q '^OutputMethod = SDL$' ../zelda3.ini
